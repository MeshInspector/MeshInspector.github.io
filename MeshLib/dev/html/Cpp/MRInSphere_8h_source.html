75template <typename T>
76class InSphereTester
77{
78 static_assert( std::is_floating_point_v<T> );
79public:
85 bool reset( const Vector3<T> & va, const Vector3<T> & vb, const Vector3<T> & vc, T sqRadius )
86 {
87 a = va;
88 rSq = sqRadius;
89 E = -1;
90 u = vb - va;
91 v = vc - va;
92
95 const T uu = u.lengthSq();
96 const T vv = v.lengthSq();
97 if ( uu > 4 * rSq || vv > 4 * rSq || ( v - u ).lengthSq() > 4 * rSq )
98 return false;
99
100 w = cross( u, v );
101 W = w.lengthSq();
102 if ( W <= 0 )
103 return false;
104
106 const T uv = dot( u, v );
107 M = ( vv * ( uu - uv ) ) * u + ( uu * ( vv - uv ) ) * v;
108
110 E = 4 * rSq * W * W - M.lengthSq();
111 return E >= 0;
112 }
113
118 void flip()
119 {
120 assert( E >= 0 );
121 std::swap( u, v );
122 w = -w;
123 }
124
127 [[nodiscard]] InSphereResult operator()( const Vector3<T> & d ) const
128 {
129 assert( E >= 0 );
130 const auto q = d - a;
131
133 const T qq = q.lengthSq();
134 if ( qq > 4 * rSq )
135 return InSphereResult::Outside;
136
137 const T A = W * qq - dot( q, M );
138 const T t = dot( q, w );
139
141 if ( A < 0 && t >= 0 )
142 return InSphereResult::Inside;
143 if ( A >= 0 && t <= 0 )
144 return ( A == 0 && ( t == 0 || E == 0 ) ) ? InSphereResult::OnSphere : InSphereResult::Outside;
145 const T lhs = A * A * W;
146 const T rhs = E * t * t;
147 if ( lhs == rhs )
148 return InSphereResult::OnSphere;
149 return ( A < 0 ) == ( lhs > rhs ) ? InSphereResult::Inside : InSphereResult::Outside;
150 }
151
152private:
153 Vector3<T> a;
154 Vector3<T> u, v;
155 Vector3<T> w;
156 T W = 0;
157 Vector3<T> M;
158 T E = -1;
159 T rSq = 0;
160};
