644void MeshTopology::forEachVertex( const MeshTriPoint & p, T && callback ) const
645{
646 if ( auto v = p.inVertex( *this ) )
647 {
648 callback( v );

664template<typename T>
665void MeshTopology::flipEdgesIn( const EdgeId e0, T && flipNeeded )
666{
667 EdgeId e = e0;
668 for (;;)
669 {
670 auto testEdge = prev( e.sym() );
671 if ( left( testEdge ) && right( testEdge ) && flipNeeded( testEdge ) )
672 flipEdge( testEdge );
673 else
674 {
675 e = next( e );
676 if ( e == e0 )
677 break;
678 }
679 }
680}

682template<typename T>
683void MeshTopology::flipEdgesOut( EdgeId e0, T && flipNeeded )
684{
685 EdgeId e = e0;
686 for (;;)
687 {
688 if ( left( e ) && right( e ) && flipNeeded( e ) )
689 {
690 e0 = next( e );
691 flipEdge( e );
692 e = e0;
693 }
694 else
695 {
696 e = next( e );
697 if ( e == e0 )
698 break;
699 }

705[[nodiscard]] Vector<T, I> rearrangeVectorByMap( const Vector<T, I>& oldVector, const BMap<I, I>& map )

707 Vector<T, I> newVector;
708 newVector.resize( map.tsize );
709

715 if ( newV.valid() )
