195template <typename C, typename D, D M>
196void SparsePolynomial<C,D,M>::mergeTerms_()
197{
198 size_t out = 0;
199 for ( size_t i = 0; i < terms_.size(); )
200 {
201 auto deg = terms_[i].first;
202 auto cf = std::move( terms_[i].second );
203 for ( ++i; i < terms_.size() && terms_[i].first == deg; ++i )
204 cf += terms_[i].second;
205 if ( cf != 0 )
206 terms_[out++] = { deg, std::move( cf ) };
207 }
208 terms_.resize( out );
209}

224template <typename C, typename D, D M>
225SparsePolynomial<C,D,M>& SparsePolynomial<C,D,M>::operator -=( const SparsePolynomial& b )
226{
227 std::vector<Term> res;
228 res.reserve( terms_.size() + b.terms_.size() );
229 auto itA = terms_.begin();
230 auto itB = b.terms_.begin();
231 while ( itA != terms_.end() && itB != b.terms_.end() )
232 {
233 if ( itB->first < itA->first )
234 {
235 res.emplace_back( itB->first, -itB->second );
236 ++itB;
237 }
238 else
239 {
240 res.push_back( std::move( *itA ) );
241 ++itA;
242 }
243 }
244 for ( ; itA != terms_.end(); ++itA )
245 res.push_back( std::move( *itA ) );
246 for ( ; itB != b.terms_.end(); ++itB )
247 res.emplace_back( itB->first, -itB->second );
248 terms_ = std::move( res );
249 mergeTerms_();
250 return * this;
251}

253template <typename T, typename C, typename D, D M>
254[[nodiscard]] SparsePolynomialProduct<T,D,M> mulAs( const SparsePolynomial<C,D,M>& a, const SparsePolynomial<C,D,M>& b )
255{
256 using Res = SparsePolynomialProduct<T,D,M>;
257 using Term = typename Res::Term;
258 std::vector<Term> res;
259 res.reserve( a.terms_.size() * b.terms_.size() );
260 for ( const auto & [degA, cfA] : a.terms_ )
261 {
262 assert( cfA != 0 );
263 for ( const auto & [degB, cfB] : b.terms_ )
264 {
265 assert( cfB != 0 );
266 const auto deg = degA + degB;
267 if ( deg > M )
268 break;
270 if constexpr ( std::is_same_v<T, C> )
271 res.emplace_back( deg, cfA * cfB );
272 else
273 res.emplace_back( deg, T( cfA ) * T( cfB ) );
274 }
275 }
276 return Res::fromUnsortedTerms( std::move( res ) );
277}

279template <typename TA, typename TB, typename CA, typename CB, typename D, D M>
280std::pair<D, int> lowestTermOfSumOfProducts( const SignedProduct<SparsePolynomial<CA,D,M>, SparsePolynomial<CB,D,M>>* products, int numProducts,
281 std::type_identity_t<D> from )
282{
285 struct RowTerm
286 {
287 D deg;
288 int p, i, j;
289 };
290 auto greater = []( const RowTerm & x, const RowTerm & y ) { return x.deg > y.deg; };
291 std::vector<RowTerm> heap;
292 size_t numRows = 0;
293 for ( int p = 0; p < numProducts; ++p )
294 numRows += products[p].a->get().size();
295 heap.reserve( numRows );
296 for ( int p = 0; p < numProducts; ++p )
297 {
298 const auto & f = products[p].a->get();
299 const auto & g = products[p].b->get();
300 if ( g.empty() )
301 continue;
303 const int jFrom = from > 0 ? int( std::lower_bound( g.begin(), g.end(), from, []( const auto & t, D d ) { return t.first < d; } ) - g.begin() ) : 0;
304 for ( int i = 0; i < (int)f.size(); ++i )
305 {
306 const int j = f[i].first < from ? jFrom : 0;
307 if ( j == (int)g.size() )
308 continue;
311 if ( const D deg = f[i].first + g[j].first; deg <= M )
312 heap.push_back( { deg, p, i, j } );
313 else if ( j == 0 )
314 break;
315 }
316 }
317 std::make_heap( heap.begin(), heap.end(), greater );
318
320 const auto mul = []( const CA & x, const CB & y )
321 {
322 if constexpr ( std::is_same_v<TA, CA> && std::is_same_v<TB, CB> )
323 return x * y;
324 else if constexpr ( std::is_same_v<TA, CA> )
325 return x * TB( y );
326 else if constexpr ( std::is_same_v<TB, CB> )
327 return TA( x ) * y;
328 else
329 return TA( x ) * TB( y );
330 };
331 while ( !heap.empty() )
332 {
333 const auto deg = heap.front().deg;
334 decltype( mul( std::declval<CA>(), std::declval<CB>() ) ) coeff{};
335 do
336 {
337 std::pop_heap( heap.begin(), heap.end(), greater );
338 auto r = heap.back();
339 heap.pop_back();
340 const auto & f = products[r.p].a->get();
341 const auto & g = products[r.p].b->get();
342 const auto prod = mul( f[r.i].second, g[r.j].second );
343 if ( products[r.p].negative )
344 coeff -= prod;
345 else
346 coeff += prod;
347 if ( ++r.j < (int)g.size() && ( r.deg = f[r.i].first + g[r.j].first ) <= M )
348 {
349 heap.push_back( r );
350 std::push_heap( heap.begin(), heap.end(), greater );
351 }
352 }
353 while ( !heap.empty() && heap.front().deg == deg );
354 if ( coeff > 0 )
355 return { deg, 1 };
356 if ( coeff < 0 )
357 return { deg, -1 };
358 }
359 return {};
360}
