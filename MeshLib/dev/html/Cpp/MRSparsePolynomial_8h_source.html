189template <typename C, typename D, D M>
190[[nodiscard]] SparsePolynomial<C,D,M> operator *( const SparsePolynomial<C,D,M>& a, const SparsePolynomial<C,D,M>& b )
191{
192 using Term = typename SparsePolynomial<C,D,M>::Term;
193 std::vector<Term> res;
194 res.reserve( a.terms_.size() * b.terms_.size() );
195 for ( const auto & [degA, cfA] : a.terms_ )
196 {
197 assert( cfA != 0 );
198 for ( const auto & [degB, cfB] : b.terms_ )
199 {
200 assert( cfB != 0 );
201 const auto deg = degA + degB;
202 if ( deg > M )
203 break;
204 res.emplace_back( deg, cfA * cfB );
205 }
206 }
207 std::sort( res.begin(), res.end(),
208 []( const Term & x, const Term & y ) { return x.first < y.first; } );
209 SparsePolynomial<C,D,M> r;
210 r.terms_ = std::move( res );
211 r.mergeTerms_();
212 return r;
213}
