56bool rayBoxIntersect( const Box3<T>& box, const RayOrigin<T> & rayOrigin, T & t0, T & t1, const IntersectionPrecomputes<T>& prec )
57{
58 assert( box.valid() );
59
60 #if defined(__x86_64__) || defined(_M_X64)
61 if constexpr (std::is_same_v<T, float>)
62 {
64 static_assert( sizeof( Box3f ) == 6 * sizeof( float ) );
65 const float * const c = &box.min.x;
66 __m128 l = _mm_loadu_ps( c );
67 __m128 h = _mm_loadu_ps( c + 2 );
68 __m128 r = _mm_shuffle_ps( h, h, _MM_SHUFFLE( 3, 3, 2, 1 ) );
69
70 l = _mm_mul_ps( _mm_sub_ps( l, rayOrigin.p ), prec.invDir );
71 r = _mm_mul_ps( _mm_sub_ps( r, rayOrigin.p ), prec.invDir );
72
73 __m128 a = _mm_min_ps( l, r );
74 __m128 b = _mm_max_ps( l, r );
75
77 __m128 az = _mm_shuffle_ps( a, a, _MM_SHUFFLE( 2, 2, 2, 2 ) );
78 __m128 ay = _mm_shuffle_ps( a, a, _MM_SHUFFLE( 1, 1, 1, 1 ) );
79 t0 = _mm_cvtss_f32( _mm_max_ss( _mm_max_ss( _mm_max_ss( a, az ), ay ), _mm_load_ss( &t0 ) ) );
80
81 __m128 bz = _mm_shuffle_ps( b, b, _MM_SHUFFLE( 2, 2, 2, 2 ) );
82 __m128 by = _mm_shuffle_ps( b, b, _MM_SHUFFLE( 1, 1, 1, 1 ) );
83 t1 = _mm_cvtss_f32( _mm_min_ss( _mm_min_ss( _mm_min_ss( b, bz ), by ), _mm_load_ss( &t1 ) ) );

86 }
87 else
88 #elif defined(__aarch64__) || defined(_M_ARM64)
89 if constexpr (std::is_same_v<T, float>)

103 t1 = vminvq_f32( vmaxq_f32( l, r ) );

106 }
107 else

128 v128_t bz = wasm_i32x4_shuffle( b, b, 2, 2, 2, 2 );

132 t1 = std::min( t1, wasm_f32x4_extract_lane( wasm_f32x4_pmin( wasm_f32x4_pmin( b, bz ), by ), 0 ) );

135 }

144 t1 = std::min( (box[sign.x].x - rayOrigin.p.x) * prec.invDir.x, t1 );
145 t0 = std::max( (box[1 - sign.x].x - rayOrigin.p.x) * prec.invDir.x, t0 );
