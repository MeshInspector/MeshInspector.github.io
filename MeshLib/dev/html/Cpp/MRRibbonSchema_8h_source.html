182template<typename T>
183class RibbonMenuItemCall
184{
185 static_assert( std::is_base_of_v<RibbonMenuItem, T> );
186public:
187 template<typename F>
188 RibbonMenuItemCall( F f, std::function<void(std::shared_ptr<T>)> g ) : g_( std::move( g ) )
189 {
190 const auto& items = RibbonSchemaHolder::schema().items;
191 for ( const auto& item : items )
192 {
193 auto plugin = std::dynamic_pointer_cast< T >( item.second.item );
194 if ( !plugin )
195 continue;
196 f( plugin_ = std::move( plugin ) );
197 break;
198 }
199 }
200 ~RibbonMenuItemCall()
201 {
202 if ( g_ && plugin_ )
203 g_( plugin_ );
204 }
205private:
206 std::shared_ptr<T> plugin_;
207 std::function<void(std::shared_ptr<T>)> g_;
208};
