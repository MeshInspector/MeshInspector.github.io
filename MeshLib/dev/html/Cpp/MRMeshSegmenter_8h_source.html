87template<typename Policy>
88void MeshSegmenter<Policy>::constructGraph_()
89{
90 MR_TIMER;
91
95
96 graphEdgeData_.resize( topology_.undirectedEdgeSize() );
97 Graph::EndsPerEdge ends( topology_.undirectedEdgeSize() );
98 Graph::EdgeBitSet validGraphEdges;
99 static_cast<BitSet&>( validGraphEdges ) = topology_.findNotLoneUndirectedEdges();
100 BitSetParallelFor( validGraphEdges, [&]( Graph::EdgeId ge )
101 {
102 UndirectedEdgeId ue( (int)ge );
103
104 Graph::EndVertices vv;
105 vv.v0 = Graph::VertId( (int)topology_.left( ue ) );
106 vv.v1 = Graph::VertId( (int)topology_.right( ue ) );
107 if ( !vv.v0 || !vv.v1 )
108 {
109 validGraphEdges.reset( ge );
110 return;
111 }
112
113 assert( vv.v0 != vv.v1 );
114 if ( vv.v1 < vv.v0 )
115 std::swap( vv.v0, vv.v1 );
116 ends[ge] = vv;
117 graphEdgeData_[ge] = policy_.bdFromEdge( ue );
118 } );
119
120 graphVertData_.resize( topology_.faceSize() );
121 Graph::NeighboursPerVertex neis( topology_.faceSize() );
122 Graph::VertBitSet validGraphVerts;
123 static_cast<BitSet&>( validGraphVerts ) = topology_.getValidFaces();
124 BitSetParallelFor( validGraphVerts, [&]( Graph::VertId gv )
125 {
126 FaceId f( (int)gv );
127 Graph::Neighbours n;
128 n.reserve( topology_.getFaceDegree( f ) );
129 for ( auto e : leftRing( topology_, f ) )
130 {
131 Graph::EdgeId ge( int( e.undirected() ) );
132 assert( topology_.left( e ) == f );
133 if ( topology_.right( e ) )
134 n.push_back( ge );
135 }
136 graphVertData_[gv] = policy_.segmentFromTri( f );
137 std::sort( n.begin(), n.end() );
138 neis[gv] = std::move( n );
139 } );
140 graph_.construct( std::move( neis ), std::move( validGraphVerts ), std::move( ends ), std::move( validGraphEdges ) );
141}

159template<typename Policy>
160std::optional<Graph::EndVertices> MeshSegmenter<Policy>::mergeNext_()
161{
162 const auto [ge, penalty] = heap_.top();
163 if ( penalty == prohibitMergePenalty )
164 return std::nullopt;
165 assert( graph_.valid( ge ) );
166 assert( penalty == penalty_( ge ) );
167 heap_.setSmallerValue( ge, prohibitMergePenalty );
168
169 auto ends = graph_.ends( ge );
170 graphVertData_[ends.v0] += graphVertData_[ends.v1];
171 graph_.merge( ends.v0, ends.v1, [&]( Graph::EdgeId remnant, Graph::EdgeId dead )
172 {
173 graphEdgeData_[remnant] += graphEdgeData_[dead];
174 heap_.setSmallerValue( dead, prohibitMergePenalty );
175 } );
176
177 for ( auto ne : graph_.neighbours( ends.v0 ) )
178 heap_.setValue( ne, penalty_( ne ) );
179
180 return ends;
181}

185{
186 MR_TIMER;
187 GroupOrder res;

191 {
192 res.push_back( { FaceId( int( vv->v0 ) ), FaceId( int( vv->v1 ) ) } );
193 if ( !reportProgress( progress, [&]{ return float( iter ) / numIters; }, ++iter, 1024 ) )
194 return unexpectedOperationCanceled();
195 }
