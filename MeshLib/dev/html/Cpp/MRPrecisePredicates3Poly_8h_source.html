29[[nodiscard]] MR_FORCE_INLINE FastInt128 orient3dVolume( const Vector3i & a, const Vector3i & b, const Vector3i & c, const Vector3i & d )
30{
31 const Vector3i64 x( a - d );
32 const Vector3i64 y( b - d );
33 const Vector3i64 z( c - d );
34
35 return
36 Int64Mul128( x.x ) * Int64Mul128( y.y * z.z - y.z * z.y )
37 - Int64Mul128( x.y ) * Int64Mul128( y.x * z.z - y.z * z.x )
38 + Int64Mul128( x.z ) * Int64Mul128( y.x * z.y - y.y * z.x );
39}

64MR_FORCE_INLINE int getPointDegrees( const PreciseVertCoords * vs, const std::int64_t * radices, int n, PointDegree * res,
65 std::int64_t * places = nullptr )
66{
67 struct VertN
68 {
69 VertId v;
70 int n = 0;
71 };
72 assert( n > 0 && n <= cMaxPerturbedPoints );
73 std::array<VertN, cMaxPerturbedPoints> as;
74 for ( int i = 0; i < n; ++i )
75 as[i] = { vs[i].id, i };
76 std::sort( as.begin(), as.begin() + n, []( const auto & a, const auto & b ) { return a.v < b.v; } );
77
78 std::int64_t d = 1;
79 int numIds = 1;
80 if ( places )
81 places[0] = d;
82 for ( int i = 0; i < n; ++i )
83 {
84 const auto k = as[i].n;
85 res[k] = { vs[k].pt, d };
86 if ( i + 1 == n )
87 break;
88 const auto r = radices[k];
89 if ( as[i].v < as[i+1].v )
90 {
91 assert( d <= std::numeric_limits<std::int64_t>::max() / r );
92 d *= r;
93 if ( places )
94 places[numIds] = d;
95 ++numIds;
96 }
97 else
98 assert( r == radices[as[i+1].n] );
99 }
100 return numIds;
101}

168[[nodiscard]] VolumePoly<M> orient3dPoly( const PointDegree & a, const PointDegree & b, const PointDegree & c, const PointDegree & d )
169{
172 struct Element
173 {
174 typename VolumePoly<M>::Term terms[3];
175 int numTerms = 0;
176 };
177 Element m[3][3];
178 const PointDegree * rows[3] = { &a, &b, &c };
179 for ( int r = 0; r < 3; ++r )
180 for ( int col = 0; col < 3; ++col )
181 {
182 auto & e = m[r][col];
183 if ( const std::int64_t delta = std::int64_t( rows[r]->pt[col] ) - d.pt[col] )
184 e.terms[e.numTerms++] = { 0, delta };
185 if ( rows[r]->d )
186 e.terms[e.numTerms++] = { rows[r]->degree( col ), 1 };
187 if ( d.d )
188 e.terms[e.numTerms++] = { d.degree( col ), -1 };
189 }
190
193 constexpr int perms[6][4] = { { 0, 1, 2, 1 }, { 1, 2, 0, 1 }, { 2, 0, 1, 1 }, { 0, 2, 1, -1 }, { 2, 1, 0, -1 }, { 1, 0, 2, -1 } };
194 std::vector<typename VolumePoly<M>::Term> terms;
195 terms.reserve( 6 * 26 );
196 for ( const auto & p : perms )
197 {
198 const Element & e0 = m[0][p[0]];
199 const Element & e1 = m[1][p[1]];
200 const Element & e2 = m[2][p[2]];
201 for ( int k0 = 0; k0 < e0.numTerms; ++k0 )
202 for ( int k1 = 0; k1 < e1.numTerms; ++k1 )
203 {
205 const std::int64_t c01 = p[3] * std::int64_t( e0.terms[k0].second ) * std::int64_t( e1.terms[k1].second );
206 const std::int64_t d01 = e0.terms[k0].first + e1.terms[k1].first;
207 const int last2 = d.d && k0 + 1 == e0.numTerms && k1 + 1 == e1.numTerms ? e2.numTerms - 1 : e2.numTerms;
208 for ( int k2 = 0; k2 < last2; ++k2 )
209 if ( const auto deg = d01 + e2.terms[k2].first; deg <= M )
210 terms.emplace_back( deg, Int64Mul128( c01 ) * Int64Mul128( std::int64_t( e2.terms[k2].second ) ) );
211 }
212 }
213 return VolumePoly<M>::fromUnsortedTerms( std::move( terms ) );
214}
