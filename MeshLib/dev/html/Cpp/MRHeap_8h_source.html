157{
158 const size_t pos = size_t( id2PosInHeap_[ elemId ] );
159 assert( heap_[pos].id == elemId );
160 id2PosInHeap_[ elemId ] = InvalidPos;

201template <typename T, typename I, typename P>
202void Heap<T, I, P>::lift_( size_t pos, I elemId )
203{
204 while ( pos > 0 )
205 {
206 size_t parentPos = ( pos - 1 ) / 2;
207 if ( !( less_( parentPos, pos ) ) )
208 break;
209 auto parentId = heap_[parentPos].id;
210 assert( size_t( id2PosInHeap_[parentId] ) == parentPos );
211 std::swap( heap_[parentPos], heap_[pos] );
212 std::swap( parentPos, pos );
213 id2PosInHeap_[parentId] = SizeType( parentPos );
214 }
215 id2PosInHeap_[elemId] = SizeType( pos );
216}

218template <typename T, typename I, typename P>

222 assert( heap_[pos].id == elemId );
223 assert( !( pred_( heap_[pos].val, newVal ) ) );
224 heap_[pos].val = newVal;
225 for (;;)
226 {
227 size_t child1Pos = 2 * pos + 1;

230 auto child1Id = heap_[child1Pos].id;
