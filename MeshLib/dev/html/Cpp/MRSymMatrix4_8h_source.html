181template <typename T>
182Vector4<T> SymMatrix4<T>::eigens( Matrix4<T> * eigenvectors ) const MR_REQUIRES_IF_SUPPORTED( std::is_floating_point_v<T> )
183{
186 T a[4][4] =
187 {
188 { xx, xy, xz, xw },
189 { xy, yy, yz, yw },
190 { xz, yz, zz, zw },
191 { xw, yw, zw, ww }
192 };
193 Matrix4<T> v;
194 constexpr T eps2 = std::numeric_limits<T>::epsilon() * std::numeric_limits<T>::epsilon() / 64;
196 constexpr int P[6] = { 0, 0, 0, 1, 1, 2 }, Q[6] = { 1, 2, 3, 2, 3, 3 }, R1[6] = { 2, 1, 1, 0, 0, 0 }, R2[6] = { 3, 3, 2, 3, 2, 1 };
197 for ( int sweep = 0; sweep < 32; ++sweep )
198 {
199 bool rotated = false;
200 for ( int i = 0; i < 6; ++i )
201 {
202 const int p = P[i], q = Q[i];
203 const T apq = a[p][q];
205 if ( apq * apq <= eps2 * std::abs( a[p][p] * a[q][q] ) )
206 {
207 a[p][q] = a[q][p] = 0;
208 continue;
209 }
210 rotated = true;
212 const T d = a[q][q] - a[p][p];
213 T t, c;
214 if ( std::abs( d ) > T( 1 << 28 ) * std::abs( apq ) )
215 {
217 t = apq / d;
218 c = 1;
219 }
220 else
221 {
223 const T h = std::sqrt( d * d + 4 * apq * apq );
224 const T u = std::abs( d ) + h;
225 t = 2 * apq / std::copysign( u, d );
226 c = std::sqrt( u / ( 2 * h ) );
227 }
228 const T s = t * c;
229 a[p][p] -= t * apq;
230 a[q][q] += t * apq;
231 a[p][q] = a[q][p] = 0;
232 for ( int k : { R1[i], R2[i] } )
233 {
234 const T akp = a[k][p], akq = a[k][q];
235 a[k][p] = a[p][k] = c * akp - s * akq;
236 a[k][q] = a[q][k] = s * akp + c * akq;
237 }
238 const Vector4<T> vp = v[p], vq = v[q];
239 v[p] = c * vp - s * vq;
240 v[q] = s * vp + c * vq;
241 }
242 if ( !rotated )
243 break;
244 }
245 int order[4] = { 0, 1, 2, 3 };
246 std::sort( order, order + 4, [&]( int i, int j ) { return a[i][i] < a[j][j]; } );
247 Vector4<T> eig;
248 for ( int i = 0; i < 4; ++i )
249 eig[i] = a[order[i]][order[i]];
250 if ( eigenvectors )
251 *eigenvectors = Matrix4<T>( v[order[0]], v[order[1]], v[order[2]], v[order[3]] );
252 return eig;
253}

255template <typename T>
256SymMatrix4<T> SymMatrix4<T>::pseudoinverse( T tol, int * rank ) const MR_REQUIRES_IF_SUPPORTED( std::is_floating_point_v<T> )
257{
258 SymMatrix4<T> res;
259 Matrix4<T> eigenvectors;
260 const auto eigenvalues = eigens( &eigenvectors );
261 const auto threshold = std::max( std::abs( eigenvalues[0] ), std::abs( eigenvalues[3] ) ) * tol;
262 int myRank = 0;
263 for ( int i = 0; i < 4; ++i )
264 {
265 if ( std::abs( eigenvalues[i] ) <= threshold )
266 continue;
267 res += outerSquare( 1 / eigenvalues[i], eigenvectors[i] );
268 ++myRank;
269 }
270 if ( rank )
271 *rank = myRank;
272 return res;
273}

275template <typename T>
276Vector4<T> SymMatrix4<T>::solveLDLT( const Vector4<T> & b ) const MR_REQUIRES_IF_SUPPORTED( std::is_floating_point_v<T> )
277{
280 const auto inv = []( T d ) { return d > 0 ? 1 / d : T( 0 ); };
281 const T i0 = inv( xx );
282 const T l10 = xy * i0, l20 = xz * i0, l30 = xw * i0;
283 const T i1 = inv( yy - l10 * xy );
284 const T e21 = yz - l20 * xy, e31 = yw - l30 * xy;
285 const T l21 = e21 * i1, l31 = e31 * i1;
286 const T i2 = inv( zz - l20 * xz - l21 * e21 );
287 const T e32 = zw - l30 * xz - l31 * e21;
288 const T l32 = e32 * i2;
289 const T i3 = inv( ww - l30 * xw - l31 * e31 - l32 * e32 );
291 const T y0 = b.x;
292 const T y1 = b.y - l10 * y0;
293 const T y2 = b.z - l20 * y0 - l21 * y1;
294 const T y3 = b.w - l30 * y0 - l31 * y1 - l32 * y2;
296 Vector4<T> x;
297 x.w = y3 * i3;
298 x.z = y2 * i2 - l32 * x.w;
299 x.y = y1 * i1 - l21 * x.z - l31 * x.w;
300 x.x = y0 * i0 - l10 * x.y - l20 * x.z - l30 * x.w;
301 return x;
302}
