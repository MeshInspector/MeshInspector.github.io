176template <typename T>
177Vector4<T> SymMatrix4<T>::eigens( Matrix4<T> * eigenvectors ) const MR_REQUIRES_IF_SUPPORTED( std::is_floating_point_v<T> )
178{
181 T a[4][4] =
182 {
183 { xx, xy, xz, xw },
184 { xy, yy, yz, yw },
185 { xz, yz, zz, zw },
186 { xw, yw, zw, ww }
187 };
188 Matrix4<T> v;
189 for ( int sweep = 0; sweep < 32; ++sweep )
190 {
191 bool rotated = false;
192 for ( int p = 0; p < 3; ++p )
193 for ( int q = p + 1; q < 4; ++q )
194 {
195 const T apq = a[p][q];
197 if ( apq == 0 || std::abs( apq ) <= std::numeric_limits<T>::epsilon() * std::sqrt( std::abs( a[p][p] * a[q][q] ) ) / 8 )
198 {
199 a[p][q] = a[q][p] = 0;
200 continue;
201 }
202 rotated = true;
203 const T theta = ( a[q][q] - a[p][p] ) / ( 2 * apq );
204 const T t = ( theta >= 0 ? T( 1 ) : T( -1 ) ) / ( std::abs( theta ) + std::sqrt( theta * theta + 1 ) );
205 const T c = 1 / std::sqrt( t * t + 1 );
206 const T s = t * c;
207 a[p][p] -= t * apq;
208 a[q][q] += t * apq;
209 a[p][q] = a[q][p] = 0;
210 for ( int k = 0; k < 4; ++k )
211 {
212 if ( k != p && k != q )
213 {
214 const T akp = a[k][p], akq = a[k][q];
215 a[k][p] = a[p][k] = c * akp - s * akq;
216 a[k][q] = a[q][k] = s * akp + c * akq;
217 }
218 const T vpk = v[p][k], vqk = v[q][k];
219 v[p][k] = c * vpk - s * vqk;
220 v[q][k] = s * vpk + c * vqk;
221 }
222 }
223 if ( !rotated )
224 break;
225 }
226 int order[4] = { 0, 1, 2, 3 };
227 std::sort( order, order + 4, [&]( int i, int j ) { return a[i][i] < a[j][j]; } );
228 Vector4<T> eig;
229 for ( int i = 0; i < 4; ++i )
230 eig[i] = a[order[i]][order[i]];
231 if ( eigenvectors )
232 *eigenvectors = Matrix4<T>( v[order[0]], v[order[1]], v[order[2]], v[order[3]] );
233 return eig;
234}

236template <typename T>
237SymMatrix4<T> SymMatrix4<T>::pseudoinverse( T tol, int * rank ) const MR_REQUIRES_IF_SUPPORTED( std::is_floating_point_v<T> )
238{
239 SymMatrix4<T> res;
240 Matrix4<T> eigenvectors;
241 const auto eigenvalues = eigens( &eigenvectors );
242 const auto threshold = std::max( std::abs( eigenvalues[0] ), std::abs( eigenvalues[3] ) ) * tol;
243 int myRank = 0;
244 for ( int i = 0; i < 4; ++i )
245 {
246 if ( std::abs( eigenvalues[i] ) <= threshold )
247 continue;
248 res += outerSquare( 1 / eigenvalues[i], eigenvectors[i] );
249 ++myRank;
250 }
251 if ( rank )
252 *rank = myRank;
253 return res;
254}
