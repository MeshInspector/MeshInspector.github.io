51template <std::size_t n>
52[[nodiscard]] constexpr std::array<std::uint64_t, 2 * n> mulWords(
53 const std::array<std::uint64_t, n> & a, const std::array<std::uint64_t, n> & b ) noexcept
54{
55 std::array<std::uint64_t, 2 * n> res = {};
56 for ( std::size_t i = 0; i < n; ++i )
57 {
58 std::uint64_t carry = 0;
59 for ( std::size_t j = 0; j < n; ++j )
60 {
62 const FastUInt128 t = FastUInt128( a[i] ) * FastUInt128( b[j] ) + FastUInt128( res[i + j] ) + FastUInt128( carry );
63 res[i + j] = std::uint64_t( t );
64 carry = std::uint64_t( t >> 64 );
65 }
66 res[i + n] = carry;
67 }
68
70 if ( std::int64_t( a[n - 1] ) < 0 )
71 {
72 std::uint64_t borrow = 0;
73 for ( std::size_t i = 0; i < n; ++i )
74 res[n + i] = subBorrow64( res[n + i], b[i], borrow );
75 }
76 if ( std::int64_t( b[n - 1] ) < 0 )
77 {
78 std::uint64_t borrow = 0;
79 for ( std::size_t i = 0; i < n; ++i )
80 res[n + i] = subBorrow64( res[n + i], a[i], borrow );
81 }
82 return res;
83}
