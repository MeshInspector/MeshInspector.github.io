354template <typename V>
355inline Box<V> transformed( const Box<V> & box, const AffineXf<V> & xf )
356{
357 if ( !box.valid() )
358 return {};
361 Box<V> res;
362 for ( int i = 0; i < Box<V>::elements; ++i )
363 {
364 const auto & row = xf.A[i];
365 V lo, hi;
366 for ( int j = 0; j < Box<V>::elements; ++j )
367 {
368 const bool pos = row[j] >= 0;
369 lo[j] = pos ? box.min[j] : box.max[j];
370 hi[j] = pos ? box.max[j] : box.min[j];
371 }
372 res.min[i] = dot( row, lo ) + xf.b[i];
373 res.max[i] = dot( row, hi ) + xf.b[i];
374 }
375 return res;
376}

409template <typename V>
410inline auto findSortedBoxDims( const Box<V>& box ) -> typename VectorTraits<V>::template ChangeBaseType<int>
411{
412 constexpr auto es = Box<V>::elements;
413 auto boxDiag = box.max - box.min;
414 std::pair<float, int> ps[es];
415 for ( int i = 0; i < es; ++i )
416 ps[i] = { boxDiag[i], i };
417
419 for ( int i = 0; i + 1 < es; ++i )
420 for ( int j = i + 1; j < es; ++j )
421 if ( ps[j] < ps[i] )
422 std::swap( ps[i], ps[j] );
423
424 typename VectorTraits<V>::template ChangeBaseType<int> res( noInit );
425 for ( int i = 0; i < es; ++i )
426 res[i] = ps[i].second;
427 return res;
428}
