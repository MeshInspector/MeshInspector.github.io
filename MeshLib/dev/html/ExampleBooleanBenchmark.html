Boolean Benchmark

Measures the speed of boolean operations: subtracts from a sphere the same sphere rotated around Z-axis by an angle increasing by 0.1 degrees each time, and prints the elapsed time and the number of Booleans per second. The rotation is passed as rigidB2A, so neither mesh is modified and their AABB trees are built only once.

    {
    // the number of Booleans can be given in the command line
    const int numBooleans = argc > 1 ? std::atoi( argv[1] ) : 2000;
    // two identical spheres of radius 1

    auto subtractRotated = [&] ( int i )
    {
    const auto rigidB2A = MR::AffineXf3f::linear( MR::Matrix3f::rotation( MR::Vector3f::plusZ(), i * 0.1f * MR::PI_F / 180 ) );
    return MR::boolean( sphereA, sphereB, MR::BooleanOperation::DifferenceAB, { .rigidB2A = &rigidB2A } );
    };
    // untimed warm-up Boolean, it builds the AABB trees
    if ( auto res = subtractRotated( 1 ); !res.valid() )
    {
    std::cerr << res.errorString << std::endl;
    return 1;
    }
    const auto start = std::chrono::steady_clock::now();
    for ( int i = 1; i <= numBooleans; ++i )
    {
    if ( auto res = subtractRotated( i ); !res.valid() )
    {
    std::cerr << res.errorString << std::endl;
    return 1;
    }
    }
    const double seconds = std::chrono::duration<double>( std::chrono::steady_clock::now() - start ).count();
    std::cout << std::fixed << numBooleans << " Booleans in " << std::setprecision( 3 ) << seconds << " s, "

    }

    // Subtracts sphereB rotated around Z-axis by i * 0.1 degrees from sphereA.
    // sphereB itself is not modified, so the AABB trees of both meshes are built once and reused.
    static bool subtractRotated( const MR_Mesh* sphereA, const MR_Mesh* sphereB, int i )
    {
    MR_Vector3f plusZ = MR_Vector3f_plusZ();
    MR_Matrix3f rotation = MR_Matrix3f_rotation_float( &plusZ, i * 0.1f * 3.14159265f / 180 );
    MR_AffineXf3f rigidB2A = MR_AffineXf3f_linear( &rotation );
    MR_BooleanResult* result = MR_boolean_withXf( sphereA, sphereB, MR_BooleanOperation_DifferenceAB, &rigidB2A, NULL, MR_PassBy_DefaultArgument, NULL );
    bool valid = MR_BooleanResult_valid( result );
    if ( !valid )
    fprintf( stderr, "Failed to perform boolean: %s\n", MR_std_string_data( MR_BooleanResult_Get_errorString( result ) ) );
    MR_BooleanResult_Destroy( result );
    return valid;
    }

    // Subtracts sphere_b rotated around Z-axis by i * 0.1 degrees from sphere_a.
    // sphere_b itself is not modified, so the AABB trees of both meshes are built once and reused.
    fn subtract_rotated(sphere_a: &Mesh, sphere_b: &Mesh, i: i32) -> Result<(), String>
    {
    let rigid_b2a = AffineXf3f::linear(&Matrix3f::rotation_float(&Vector3f::plus_z(), (i as f32 * 0.1).to_radians()));
    let result = meshlib::boolean_with_xf(sphere_a, sphere_b, BooleanOperation::DifferenceAB, Some(&rigid_b2a), None, None);
    if !result.valid()
    {
    return Err(format!("Failed to perform boolean: {}", result.error_string()));
    }
    Ok(())
    }

    {
    // The number of Booleans can be given in the command line.
