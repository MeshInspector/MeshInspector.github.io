    }

    public static class SignedDistancesExample
    {
    public static void Main(string[] args)
    {
    var refMesh = MR.MeshLoad.fromAnySupportedFormat("mesh1.ctm");
    var mesh = MR.MeshLoad.fromAnySupportedFormat("mesh2.ctm");
    // get object of VertScalars - set of distances between points of target mesh and reference mesh
    var vertDistances = MR.findSignedDistances(refMesh, mesh);
    float minDist = float.MaxValue, maxDist = float.MinValue;
    for (int i = 0; i < (int)vertDistances.size(); ++i)
    {
    float d = vertDistances[new MR.VertId(i)];
    minDist = MathF.Min(minDist, d);
    maxDist = MathF.Max(maxDist, d);
    }
    Console.WriteLine("Distance between reference mesh and the closest point of target mesh is {0}", minDist);
    Console.WriteLine("Distance between reference mesh and the farthest point of target mesh is {0}", maxDist);
    }
    }

    import createMeshLib from '@meshinspector/meshlib';
    const ml = await createMeshLib();
    using refMesh = ml.MeshLoad.fromAnySupportedFormat('mesh1.ply');
    using mesh = ml.MeshLoad.fromAnySupportedFormat('mesh2.ply');
    // distances between the points of the target mesh and the reference mesh
    // (the JS binding requires a parameters object; its defaults match the C++ overload)
    using params = new ml.MeshProjectionParameters();
    const vertDistances = ml.findSignedDistances(refMesh, mesh, params);
    // find the closest and farthest distances (a loop, not Math.min(...arr), because the
    // typed array may be large enough to overflow the call stack when spread)
    let closest = Infinity;
    let farthest = -Infinity;
    for (const d of vertDistances) {
    if (d < closest) closest = d;
    if (d > farthest) farthest = d;
    }
    console.log(`Distance between reference mesh and the closest point of target mesh is ${closest}`);
    console.log(`Distance between reference mesh and the farthest point of target mesh is ${farthest}`);
    Note
    The using declaration requires Node.js 24+ or a modern browser. On older runtimes, call .delete() on each object instead (see Memory Management in the JavaScript Setup Guide).

    use meshlib::VertId;
    fn main() -> Result<(), String>
    {
    let ref_mesh = meshlib::mesh_load::from_any_supported_format("mesh1.ctm", None).into_result()?;
    let mesh = meshlib::mesh_load::from_any_supported_format("mesh2.ctm", None).into_result()?;
    // get object of VertScalars - set of distances between points of target mesh and reference mesh
    let vert_distances = meshlib::find_signed_distances(&ref_mesh, &mesh, None, None);
    let distances: Vec<f32> = (0..vert_distances.size()).map(|i| *vert_distances.index(VertId {id_: i as i32})).collect();
    let min = distances.iter().copied().fold(f32::INFINITY, f32::min);
    let max = distances.iter().copied().fold(f32::NEG_INFINITY, f32::max);
    println!("Distance between reference mesh and the closest point of target mesh is {min}");
    println!("Distance between reference mesh and the farthest point of target mesh is {max}");
    Ok(())
    }
